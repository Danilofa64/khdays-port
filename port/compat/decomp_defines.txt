# The preprocessor definitions the game's side is compiled with -- read by
# port/CMakeLists.txt (the khdays-decomp target) and by the rewrite that
# preprocesses the same sources (port/tools/references). One per line;
# `#` starts a comment.

# the game's main; the process has its own
main=khdays_NitroMain

# The game's C library (libs/msl) defines these C runtime names; renamed, the
# game keeps its own and the rest of the process (SDL) the CRT's. (Also what an
# optimizing build would otherwise take for its intrinsics: it refuses a
# definition of strcmp or strcpy, and crashed on a call.)
strcmp=khdays_msl_strcmp
strcpy=khdays_msl_strcpy
strlen=khdays_msl_strlen
strncmp=khdays_msl_strncmp
strncpy=khdays_msl_strncpy
strtol=khdays_msl_strtol
toupper=khdays_msl_toupper
abs=khdays_msl_abs

# The game's names a Windows library exports too: on x64 a C function's
# symbol is its bare name, and the two would be one.
CreateThread=khdays_game_CreateThread
_CRT_SECURE_NO_WARNINGS
